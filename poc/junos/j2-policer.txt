# PoC J2-5(BL-275): policer(2 kpps の仮想機の上で量が測れるか)・input-list(フィルタの並び)・送信元の bogon の prefix-list と except
# 前提= j2-lo0.txt の後(lo0 の PROTECT-RE は残す。ここは中継の通信だけを見る)
FRR1:sh ping --help 2>&1 | grep -E '^\s+-(i|q|s|I) '
FRR1:sh ip addr add 10.200.0.1/32 dev lo 2>/dev/null; ip -4 addr show lo | grep inet
JUN02 configure
JUN02 set firewall policer ICMP-64K if-exceeding bandwidth-limit 64k
JUN02 set firewall policer ICMP-64K if-exceeding burst-size-limit 15k
JUN02 set firewall policer ICMP-64K then discard
JUN02 set policy-options prefix-list BOGONS 10.0.0.0/8
JUN02 set policy-options prefix-list BOGONS 100.64.0.0/10
JUN02 set policy-options prefix-list BOGONS 172.16.0.0/12
JUN02 set policy-options prefix-list BOGONS 192.168.0.0/16
JUN02 set firewall family inet filter NO-BOGONS term BOGON from source-prefix-list BOGONS
JUN02 set firewall family inet filter NO-BOGONS term BOGON then count BOGON
JUN02 set firewall family inet filter NO-BOGONS term BOGON then discard
JUN02 set firewall family inet filter NO-BOGONS term OTHER then accept
JUN02 set firewall family inet filter LIMIT-ICMP term ICMP from protocol icmp
JUN02 set firewall family inet filter LIMIT-ICMP term ICMP then policer ICMP-64K
JUN02 set firewall family inet filter LIMIT-ICMP term ICMP then count ICMP-IN
JUN02 set firewall family inet filter LIMIT-ICMP term OTHER then accept
JUN02 set interfaces et-0/0/1 unit 0 family inet filter input-list [ NO-BOGONS LIMIT-ICMP ]
JUN02 commit and-quit
JUN02 show interfaces et-0/0/1.0 extensive | match "ilter" | no-more
# 送信元 10.200.0.1(私的アドレス)の ping= 捨てられる。ISP との接続(10.0.23.2)も 10/8 に入る → BGP も捨てられる(張り直しで確かめる)
FRR1:sh ping -c 3 -W 1 -I 10.200.0.1 172.16.10.10
JUN02 clear bgp neighbor 10.0.23.2
!sleep 20
JUN02 show bgp summary | match "10.0.23" | no-more
JUN02 show firewall filter NO-BOGONS | no-more
# except= 接続の /30 を除く
JUN02 configure
JUN02 set policy-options prefix-list ISP-LINK 10.0.23.0/30
JUN02 set firewall family inet filter NO-BOGONS term BOGON from source-prefix-list ISP-LINK except
JUN02 show firewall family inet filter NO-BOGONS term BOGON
JUN02 commit and-quit
!sleep 20
JUN02 show bgp summary | match "10.0.23" | no-more
FRR1:sh ping -c 3 -W 1 -I 10.200.0.1 172.16.10.10
FRR1:sh ping -c 3 -W 1 172.16.10.10
!sleep 3
JUN02 show firewall filter NO-BOGONS | no-more
# policer: 1000 バイトの ping を 100 個/秒で 200 個(≈ 0.8 Mbps)→ 64k を超えた分が捨てられる
JUN02 clear firewall all
FRR1:sh ping -q -c 200 -i 0.01 -s 1000 172.16.10.10
!sleep 5
JUN02 show firewall filter LIMIT-ICMP | no-more
# 遅い ping(5 個/秒・56 バイト)は落ちない
JUN02 clear firewall all
FRR1:sh ping -q -c 20 -i 0.2 172.16.10.10
!sleep 5
JUN02 show firewall filter LIMIT-ICMP | no-more
JUN02 show policer | no-more
