# PoC J2-1(BL-275): JUN02 の ISP 側(et-0/0/1)の入力フィルタ。BGP・ICMP・中の網への SSH だけを通し、残りは末尾の暗黙の discard。
# tcp= FRR1 から TCP でつなぐ(rc=0 つながった・rc=124 3 秒で時間切れ= 黙って捨てられた・rc=1 すぐ拒まれた)
FRR1:sh tcp(){ local s=$EPOCHREALTIME; timeout 3 bash -c "exec 3<>/dev/tcp/$1/$2 && head -c 16 <&3" 2>&1 | head -2; echo " rc=${PIPESTATUS[0]} t=$(awk "BEGIN{printf \"%.2f\", $EPOCHREALTIME-$s}")"; }
FRR1:sh nc -h 2>&1 | head -2
JUN02 configure
JUN02 set firewall family inet filter FROM-ISP term BGP from source-address 10.0.23.2/32
JUN02 set firewall family inet filter FROM-ISP term BGP from protocol tcp
JUN02 set firewall family inet filter FROM-ISP term BGP from port bgp
JUN02 set firewall family inet filter FROM-ISP term BGP then accept
JUN02 set firewall family inet filter FROM-ISP term ICMP from protocol icmp
JUN02 set firewall family inet filter FROM-ISP term ICMP then count ICMP-IN
JUN02 set firewall family inet filter FROM-ISP term ICMP then accept
JUN02 set firewall family inet filter FROM-ISP term SSH-IN from destination-address 172.16.10.0/24
JUN02 set firewall family inet filter FROM-ISP term SSH-IN from protocol tcp
JUN02 set firewall family inet filter FROM-ISP term SSH-IN from destination-port ssh
JUN02 set firewall family inet filter FROM-ISP term SSH-IN then count SSH-IN
JUN02 set firewall family inet filter FROM-ISP term SSH-IN then accept
JUN02 set interfaces et-0/0/1 unit 0 family inet filter input FROM-ISP
JUN02 show | compare
JUN02 commit and-quit
JUN02 show firewall
# 通る: 中の網(172.16.10.1= JUN01)への SSH・ping
FRR1:sh tcp 172.16.10.1 22
FRR1:sh ping -c 5 -W 1 172.16.10.10
# 落ちる(暗黙の discard): JUN01 の lo0 への SSH(中継)・JUN02 自身のアドレスへの SSH(RE 行き。IF の入力フィルタは RE 行きにも効くか)
FRR1:sh tcp 192.168.0.1 22

JUN02 show firewall filter FROM-ISP
JUN02 show bgp summary | match "Establ|Peer"
JUN02 show interfaces et-0/0/1.0 extensive | match "Filter|filter"
# 末尾に明示の term(count＋discard)を置くと、暗黙に捨てていた数が見える
JUN02 configure
JUN02 set firewall family inet filter FROM-ISP term LAST then count DENIED
JUN02 set firewall family inet filter FROM-ISP term LAST then discard
JUN02 commit and-quit
FRR1:sh tcp 192.168.0.1 22
FRR1:sh tcp 10.0.23.1 22
JUN02 show firewall filter FROM-ISP
JUN02 show firewall counter DENIED filter FROM-ISP
